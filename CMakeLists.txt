cmake_minimum_required(VERSION 3.28.3)
project(PasswordManager VERSION 1.0.0 LANGUAGES C CXX)

option(ENABLE_COVERAGE "Enable Coverage" OFF)
option(ENABLE_LINT "Enable Lint" OFF)
option(ENABLE_ASAN "Enable AddressSanitizer (ASan)" OFF)
option(ENABLE_LSAN "Enable LeakSanitizer (LSan)" OFF)
option(ENABLE_UBSAN "Enable UndefinedBehaviorSanitizer (UBSan)" OFF)
option(ENABLE_TSAN "Enable ThreadSanitizer (TSan)" OFF)
option(ENABLE_CCACHE "Enable ccache" ON)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")

include(NoInSourceBuilds)
include(Ccache)
include(ClangTidy)
include(Config)
include(Coverage)
include(CppCheck)
include(CTest)
include(Instrumentation)
include(Rust)
include(SQLite)
include(Testing)

add_subdirectory(src)
